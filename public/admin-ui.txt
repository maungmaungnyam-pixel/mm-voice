  async function compressImage(file) {
    if (!file) return '';
    if (file.size > 12*1024*1024) throw new Error('Choose an image smaller than 12 MB.');
    const dataUrl=await new Promise((resolve,reject)=>{const reader=new FileReader();reader.onerror=()=>reject(new Error('Could not read image.'));reader.onload=()=>resolve(String(reader.result));reader.readAsDataURL(file);});
    const image=await new Promise((resolve,reject)=>{const img=new Image();img.onload=()=>resolve(img);img.onerror=()=>reject(new Error('Unsupported image.'));img.src=dataUrl;});
    let width=Math.min(image.naturalWidth,1200),height=Math.round(image.naturalHeight*width/image.naturalWidth); const canvas=document.createElement('canvas');
    for(let pass=0;pass<4;pass++){canvas.width=width;canvas.height=height;const ctx=canvas.getContext('2d');ctx.fillStyle='#fff';ctx.fillRect(0,0,width,height);ctx.drawImage(image,0,0,width,height);const output=canvas.toDataURL('image/jpeg',Math.max(.48,.78-pass*.1));if(output.length<=240000)return output;width=Math.round(width*.78);height=Math.round(height*.78);}
    throw new Error('Image is too detailed. Choose a smaller file or paste an HTTPS image URL.');
  }
